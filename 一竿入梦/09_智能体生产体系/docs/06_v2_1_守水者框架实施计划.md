function Assert-StringArray {
  param($Value, [string]$Label, [bool]$RequiredNonEmpty)
  if ($null -eq $Value) {
    if ($RequiredNonEmpty) { throw "$Label 不得为 null" }
    return
  }
  if (-not ($Value -is [array])) { throw "$Label 必须是数组" }
  $items = @($Value)
  if ($RequiredNonEmpty -and $items.Count -eq 0) { throw "$Label 必须是非空数组" }
  foreach ($item in $items) {
    if (-not ($item -is [string]) -or [string]::IsNullOrWhiteSpace($item)) { throw "$Label 的元素必须是非空字符串" }
  }
}

function Assert-StringArrayType {
  param($Value, [string]$Label)
  if (-not ($Value -is [array])) { throw "$Label 必须是数组" }
  foreach ($item in @($Value)) {
    if (-not ($item -is [string]) -or [string]::IsNullOrWhiteSpace($item)) { throw "$Label 的元素必须是非空字符串" }
  }
}

function Assert-DateValue {
  param($Value, [string]$Label, [bool]$Required)
  if ($null -eq $Value) {
    if ($Required) { throw "$Label 不得为 null" }
    return
  }
  $parsed = [datetime]::MinValue
  if (-not ($Value -is [string]) -or -not [datetime]::TryParseExact([string]$Value, "yyyy-MM-dd", [Globalization.CultureInfo]::InvariantCulture, [Globalization.DateTimeStyles]::None, [ref]$parsed)) {
    throw "$Label 必须匹配合法 YYYY-MM-DD"
  }
}

function Assert-RecordMode {
  param($Record, [string]$Label)

  $names = @($Record.PSObject.Properties | ForEach-Object { $_.Name })
  $statusProperty = $Record.PSObject.Properties["verification_status"]

  $status = [string]$statusProperty.Value
  if ($allowedStatus -notcontains $status) { throw "$Label 含非法 verification_status" }

  $nonProductionProperty = $Record.PSObject.Properties["non_production"]
  $draftProperty = $Record.PSObject.Properties["draft_output_type"]

  if ($status -eq "已核验") {
    foreach ($field in $requiredMetadata) {
      if ($names -notcontains $field) { throw "$Label 正式记录缺少 $field" }
    }
    Assert-StringArray $Record.verification_scope "$Label verification_scope" $true

    return
  }

  if ($researchStatus -contains $status) {

  }

  throw "$Label verification_status 不符合正式/研究模式"
}
